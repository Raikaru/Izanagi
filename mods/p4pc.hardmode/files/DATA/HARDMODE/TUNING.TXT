# p4pc.hardmode tuning. Every number here is our own design value, not data
# copied from the game: each multiplier scales the value the game itself
# computes for that enemy, so the player's disc stays the source of every base
# stat, reward and skill. Change these numbers to retune; bump `revision` when
# you do, so a save records which tuning its run started with.
#
# Syntax: `# comment`, `[section]`, `key = value`. Numbers are decimal, or
# hexadecimal with 0x. Unknown sections or keys, missing keys and values
# outside their limits stop the mod with an error rather than being ignored.

revision = 8

# An encounter is a boss encounter when its formation flags (the first word of
# its 0x18-byte ENCOUNT.TBL row) share a bit with this mask. In the retail
# table, 0x2 is set on every boss-type formation (the story bosses, the
# dungeon floor bosses such as Avenger Knight, the Reaper, the optional bosses
# and one tutorial battle) and on no ordinary formation. Units summoned during
# the battle inherit the encounter's class.
[classes]

# Per difficulty and class: sp, st, ma, en, ag, lu scale the enemy's maximum SP
# and its five stats (St, Ma, En, Ag, Lu); a boss's hp its maximum HP (its
# starting HP/SP follow). exp and yen scale its battle rewards. damage scales
# the HP damage of every hit the enemy deals. An ordinary enemy's hp, damage
# and exp come from the story epoch below instead (there is no ordinary.exp
# key). Results are rounded to the nearest whole number and kept inside the
# game's own limits: stats 1..99, HP/SP 1..65535, rewards 0..65535. Limits:
# 0.25..4.0 for each multiplier.
#
# No difficulty makes any enemy weaker than the original game, except the
# Charming Prince (below), whose redesigned role trades HP for repeated heals,
# and Golden's Shadow Yukiko, whose Golden HP assumed the Ice weakness a
# hard-mode run removes ([enemy 258]).

# Enemy support discipline, for enemies driven by the game's own AI tables, in
# ordinary encounters only (boss encounters are never changed). The game's own
# choices, and their random draws, always happen first; this only exchanges
# them, at two levels:
# - the table group the enemy's conditions chose: a group whose only moves are
#   HP recovery, while every living ally is above skip_heal_above of its maximum
#   HP, becomes the first group without a heal; a group without a heal, while a
#   living ally is below rescue_below, becomes the first group with one;
# - the move picked inside a group that mixes heals with other moves: a heal
#   above skip_heal_above becomes the group's highest-weight other move, and
#   another move below rescue_below becomes its highest-weight heal.
# Limits: 0..1, rescue_below below skip_heal_above; rescue_below 0 turns the
# rescue off. It is off: in the battle simulator (2026-11-03, two Magical Magus,
# 20 paired seeds) the rescue let the party win 10 fights against the original
# game's 5, while skip_heal alone allowed 4.
[ai]
skip_heal_above = 0.60
rescue_below = 0.00

# Technical: a normal hit (not a weakness or a critical, whose own bonus
# stands) of the element paired here with an ailment the target already has
# deals `party` times the damage when the attacker is on the party side and
# `enemy` times when it is on the enemy side. No extra Down or turn. The stance
# panel calls TECHNICAL. Ailments: panic, fear, rage, silence; elements: fire,
# ice, electric, wind. Each element pairs with at most one ailment. Limits:
# 1..2 for each factor.
[technical]

# Boss tactics and redesigns (design 4). Counted in base turns: a One More
# action never advances a rotation.
#
# Avenger Knight: every `knight.cycle`-th base turn he visibly charges (Power
# Charge), and his next base turn is Gale Slash on the whole party.
#
# Shadow Yukiko (4.A), steered through her own script: above `yukiko.prince_at`
# of her HP her base turns repeat Agi on a random member, Double Fangs, Mind
# Charge (the warning), Burn to Ashes, Matarunda (the recovery). At or below it
# she summons the Charming Prince once, at her next base turn (his Ice weakness
# is then known to Analyze), and repeats one ordinary pick, Mind Charge, Burn to
# Ashes, Matarunda; at or below `yukiko.final_at` she alternates Agi and Double
# Fangs again between the warned bursts, with no further summon. No resummon.
# A One More after a burst is Matarunda. The Prince's own Diarama comes at most
# once per `yukiko.prince_heal_turns` of his base turns.
#
# Shadow Kanji (4.B), steered through his and his supports' scripts: above
# `kanji.formation_at` with both supports up, his base turns repeat Fatal End
# on a random member, Power Charge (the warning), Swift Strike, his attack (the
# recovery). At or below it, or once a support falls, he changes formation once
# (Tetrakarn on himself) and repeats Fanatical Spark, Power Charge, Swift
# Strike, attack; at or below `kanji.final_at` his electric and Fatal End
# pressure alternate, and his first Power Charge there primes the finisher: a
# charged Fatal End on one member instead of Swift Strike, told apart by its
# own stance-panel cue ("Kanji sizes up one of you!"). A One More after a
# burst is his attack. Nice Guy heals (Diarama) only an ally below
# `kanji.heal_below`, at most once per `kanji.heal_turns` of his base turns,
# else covers (White Wall, Heat Riser, Dekunda, Tarukaja); Tough Guy dispels
# the party's buffs (Dekaja), else attacks, Kill Rush and Rampage in turn (no
# Rampage right after Kanji hit everyone).
#
# The factors set each burst's final damage (after the charge, guard and the
# game's own variance), so Guard on the warning roughly halves it:
# knight_sweep, yukiko_burst (Burn to Ashes), kanji_strike (Swift Strike),
# kanji_spark (Fanatical Spark), kanji_finisher (the finisher's Fatal End).
# Limits: cycle 2..8, factors 0.25..2.0, HP fractions 0.05..0.95 (final_at at
# most the earlier threshold), heal turns 1..8.
# Knight calibration (2026-11-04, attentive policy that guards on a visible
# charge, typical-player entries, 80 seeds; all figures wins of 80):
# retail 74 / Hard 52 / Harder 33.
# Yukiko redesign calibration (2026-11-06, technical policy, retail rows, 80
# seeds, Hard): retail 61; Prince 0.50 / burst 0.65 won 73, 0.75 / 0.70 won 67,
# 1.00 / 0.70 won 59 (31 rounds), 0.75 / 0.80 won 58 (19.5 rounds; retail 17).

# Boss Technical windows (design 4.D), with the restart setting BOSS PLAY and
# data/windows.xml (whose status rows must name the same enemies and ailments).
# Each `enemy.<ID>` boss can suffer that one ailment (panic 0x10, fear 0x04,
# rage 0x02, silence 0x08) with one designed landing chance: the game's own
# roll is scaled to `chance` (a fraction), so the skill's own rate no longer
# matters. A landed ailment holds through the boss's next turn and is cured at
# the one after ("<boss> recovers!" on the stance panel): a window costs the
# boss exactly one turn. It opens at most once per phase (the Knight is one
# phase: one window per fight); for the rest of the phase, and on the guarded
# turns below, a roll on the boss fails and the panel says "<boss> shrugs it
# off!". There is no wear-off roll, immune-turn counter or cap.
# Outside a hard-mode run with Technical damage the boss stays immune.
# On a hard-mode run a window boss also has no elemental weakness on any
# element, on the retail or the Golden row (p4.data.session.edit at its first
# battle; Off and non-hard runs keep the original rows): a weakness gives Down
# and One More with no cap or guard and would make the window irrelevant. The
# Charming Prince (269) is no window boss and keeps his Ice weakness.
# Limits: chance 0.01..0.95.
[windows]
chance = 0.30

# Shadow Yukiko: Fear (Ice Technical) until Freeze exists.

# Shadow Kanji: Rage (Fire Technical).
